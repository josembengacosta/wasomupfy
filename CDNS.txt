        class FastDecreasingCounter {
            constructor(element, finalNumber, options = {}) {
                this.element = element;
                this.finalNumber = this.cleanNumber(finalNumber);
                this.duration = options.duration || 2000;
                this.startMultiplier = options.startMultiplier || 5;
                this.init();
            }
            
            cleanNumber(num) {
                return parseInt(num.toString().replace(/\D/g, ''));
            }
            
            formatNumber(num) {
                return new Intl.NumberFormat('pt-BR').format(num);
            }
            
            init() {
                this.startNumber = this.finalNumber * this.startMultiplier;
                this.currentNumber = this.startNumber;
                this.startTime = null;
                this.isAnimating = false;
                
                // Salva o valor original
                this.originalContent = this.element.innerHTML;
                
                this.element.innerHTML = `
                    <span class="final-value">${this.formatNumber(this.finalNumber)}</span>
                `;
                
                this.finalElement = this.element.querySelector('.final-value');
            }
            
            start() {
                if (this.isAnimating) return;
                
                this.isAnimating = true;
                this.currentNumber = this.startNumber;
                this.startTime = performance.now();
                
                // Mostra o contador temporário
                this.finalElement.style.display = 'none';
                this.tempElement = document.createElement('span');
                this.tempElement.className = 'temp-counter';
                this.tempElement.textContent = this.formatNumber(this.currentNumber);
                this.element.prepend(this.tempElement);
                
                this.animate();
            }
            
            animate() {
                const now = performance.now();
                const elapsed = now - this.startTime;
                const progress = Math.min(elapsed / this.duration, 1);
                
                // Easing out (desacelerando no final)
                const easedProgress = 1 - Math.pow(1 - progress, 3);
                
                // Calcula número atual com desaceleração
                this.currentNumber = Math.max(
                    this.finalNumber,
                    Math.floor(this.startNumber - (this.startNumber - this.finalNumber) * easedProgress)
                );
                
                // Atualiza display
                this.tempElement.textContent = this.formatNumber(this.currentNumber);
                
                if (progress < 1) {
                    requestAnimationFrame(() => this.animate());
                } else {
                    this.complete();
                }
            }
            
            complete() {
                // Remove o contador temporário
                this.tempElement.remove();
                
                // Mostra o valor final com efeito
                this.finalElement.style.display = 'inline-block';
                this.finalElement.classList.add('active');
                
                this.isAnimating = false;
            }
            
            reset() {
                if (this.tempElement) {
                    this.tempElement.remove();
                }
                this.finalElement.classList.remove('active');
                this.finalElement.style.display = 'inline-block';
                this.isAnimating = false;
            }
        }

$ls = getDB()->prepare('SELECT last_login_at, last_login_ip FROM _users_security WHERE id_users = ?');

$ua_raw         = $current_session['user_agent'] ?? '';
$browser        = 'Desconhecido';

> Só este. O `renderDashboardAlerts` já trata plano expirado, onboarding, conta bancária, etc. Os alertas específicos da página (email não verificado, plano pendente) usa o `wuAlert()` com as classes corretas:

```php
<?php
$alertColors = [
    'danger'  => ['bg' => 'rgba(239,68,68,.08)',  'border' => 'rgba(239,68,68,.25)',  'text' => '#ef4444'],
    'warning' => ['bg' => 'rgba(234,179,8,.08)',  'border' => 'rgba(234,179,8,.25)',  'text' => '#eab308'],
    'info'    => ['bg' => 'rgba(99,102,241,.08)', 'border' => 'rgba(99,102,241,.25)', 'text' => '#6366f1'],
];
if (!function_exists('wuAlert')) {
    function wuAlert(string $type, string $icon, string $message, ?array $action = null, bool $dismiss = true, string $id = ''): void {
        global $alertColors;
        $c   = $alertColors[$type] ?? $alertColors['info'];
        $eid = $id ?: ('wuA_' . md5($message));
        echo "<div id=\"{$eid}\" style=\"display:flex;align-items:flex-start;gap:10px;"
           . "background:{$c['bg']};border:1px solid {$c['border']};border-radius:12px;"
           . "padding:.75rem 1rem;font-size:.83rem;margin-bottom:.6rem;transition:opacity .3s;\">";
        echo "<i class=\"bi {$icon}\" style=\"font-size:1rem;flex-shrink:0;margin-top:2px;color:{$c['text']};\"></i>";
        echo '<span class="wu-alert-msg">' . $message;
        if ($action) echo " <a href=\"{$action['url']}\" style=\"color:{$c['text']};font-weight:700;text-decoration:underline;white-space:nowrap\">{$action['label']} &rarr;</a>";
        echo '</span>';
        if ($dismiss) echo "<button type=\"button\" class=\"wu-alert-dismiss\" aria-label=\"Fechar\" onclick=\"(function(el){el.style.opacity='0';setTimeout(function(){el.style.display='none'},300)})(document.getElementById('{$eid}'))\">&times;</button>";
        echo '</div>';
    }
}
?>
```

(function () {

        fetch('../ajax/notifications_api.php?action=count', { credentials: 'same-origin' })

            .then(data => {
                var b = document.getElementById('navNotifBadge');

                b.textContent = n > 99 ? '99+' : n;

            }).catch(function(){});

})();
